72 template< class valuetype, uint64_t count >
73 inline
74 valuetype &staticarray<valuetype,count>::operator[](uint64_t index) {
75  // I once had (semi-clever) bounds-checking code here like:
76  //
77  // if (index>=count) {
78  // return *((valuetype *)NULL);
79  // }
80  //
81  // which would successfully return a NULL/0 if valuetype was a pointer,
82  // but would throw warnings on some platforms if valuetype isn't a
83  // pointer, and would probably crash on those platforms if it managed
84  // to run.
85  //
86  // It's (apparently) conventional for the operator[] not to do any
87  // bounds checking, not even throw any exceptions, and and just attempt
88  // to access and return the requested index, possibly crashing in the
89  // process. When a program accesses via the [] opertator it's telling
90  // the compiler "trust me, I know what I'm doing".
91  //
92  // So, for now, I removed that code above and we'll just let whatever
93  // happens, happen.
94  return data[index];
95 }
