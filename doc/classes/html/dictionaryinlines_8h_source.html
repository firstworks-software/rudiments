251 inline
252 bool dictionary<keytype,valuetype>::remove(keytype key) {
253  avltreenode<dictionarynode<keytype,valuetype> *> *tnode=find(key);
254  if (tnode) {
255  if (trackinsertionorder) {
256  list.remove(tnode->getValue());
257  }
258  if (this->collection::managekeys) {
259  node_delete_value(tnode->getValue()->getKey());
260  } else if (this->collection::managearraykeys) {
261  node_delete_array_value(tnode->getValue()->getKey());
262  }
263  if (this->collection::managevalues) {
264  node_delete_value(tnode->getValue()->getValue());
265  } else if (this->collection::managearrayvalues) {
266  node_delete_array_value(tnode->getValue()->getValue());
267  }
268  delete tnode->getValue();
269  return tree.remove(tnode);
270  }
271  return false;
272 }
