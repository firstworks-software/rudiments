175 template <class valuetype>
176 inline
177 void avltree<valuetype>::insert(treenode<valuetype> *node) {
178 
179  // degenerate case
180  if (!node) {
181  return;
182  }
183 
184  if (top) {
185 
186  // insert the node, optionally replacing the top of the tree
187  insert(top,node,&top);
188 
189  // update first
190  for (first=top;
191  first->getLeftChild();
192  first=first->getLeftChild()) {}
193 
194  // update last
195  for (last=top;
196  last->getRightChild();
197  last=last->getRightChild()) {}
198  } else {
199 
200  // if there was no top node, then this is the
201  // first node inserted into the entire tree
202  top=node;
203  first=node;
204  last=node;
205  }
206 
207  // increment length
208  length++;
209 }

211 template <class valuetype>
212 inline
213 void avltree<valuetype>::insert(treenode<valuetype> *top,
214  treenode<valuetype> *node,
215  treenode<valuetype> **treetop) {
216 
217  // degenerate case
218  if (!node) {
219  return;
220  }
221 
222  // find a location to insert the node (should always be a leaf node)
223  treenode<valuetype> *location=top;
224  for (;;) {
225 
226  if (this->getComparator()->compare(node->getValue(),
227  location->getValue())<=0) {
228 
229  if (location->getLeftChild()) {
230  location=location->getLeftChild();
231  } else {
232  location->setLeftChild(node);
233  break;
234  }
235 
236  } else if (this->getComparator()->compare(
237  node->getValue(),
238  location->getValue())>0) {
239 
240  if (location->getRightChild()) {
241  location=location->getRightChild();
242  } else {
243  location->setRightChild(node);
244  break;
245  }
246  }
247  }
248 
249  node->setParent(location);
250 
251  // update heights up the tree
252  top->adjustParentHeights(node);
253 
254  // balance the tree
255  node->getParent()->balance(treetop);
256 }

261 
262  // degenerate case
263  if (!node) {
264  return NULL;
265  }
266 
267  // update first
268  if (first==node) {
269  first=node->getNext();

273  if (last==node) {
274  last=node->getPrevious();

358 template <class valuetype>
359 inline
360 treenode<valuetype> *avltree<valuetype>::find(
361  treenode<valuetype> *startnode,
362  valuetype value) {
363 
364  // descend the tree until we find the value or run off of the bottom
365  treenode<valuetype> *current=startnode;
366  while (current) {
367 
368  int32_t result=this->getComparator()->compare(
369  current->getValue(),value);
370 
371  if (result<0) {
372  current=current->getRightChild();
373  } else if (result==0) {
374  break;
375  } else if (result>0) {
376  current=current->getLeftChild();
377  }
378  }
379 
380  return current;
381 }

383 template <class valuetype>
384 inline
385 bool avltree<valuetype>::clear() {
386 
387  // start at the top
388  treenode<valuetype> *node=top;
389  while (node) {
390 
391  // go left as far as possible
392  while (node->getLeftChild()) {
393  node=node->getLeftChild();
394  }
395 
396  // go right as far as possible
397  while (node->getRightChild()) {
398  node=node->getRightChild();
399  }
400 
401  // remove this node from its parent
402  treenode<valuetype> *p=node->getParent();
403  if (p) {
404  if (p->getLeftChild()==node) {
405  p->setLeftChild(NULL);
406  } else {
407  p->setRightChild(NULL);
408  }
409  }
410 
411  // delete the value in the node
412  node_delete_value(&(node->getReference()),
413  this->getManageValues(),
414  this->getManageArrayValues());
415 
416  // delete the node itself
417  delete node;
418 
419  // continue with the parent...
420  node=p;
421  }
422 
423  // clear pointers and length
424  top=NULL;
425  first=NULL;
426  last=NULL;
427  length=0;
428 
429  return true;
430 }

435  treenode<valuetype> *node,

437  uint16_t *indentlevel,

439  bool indent) {
440 
441  // FIXME: pay attention to indent

474  writeNodeXml(out,right,"r",

476  }

614 template <class valuetype>
615 inline
616 treenode<valuetype> *avltreenode<valuetype>::getPrevious() {
617 
618  // reverse in-order, depth-first traversal...
619 
620  if (left) {
621 
622  // if we have a left child then its rightmost descendent
623  // contains the next lowest value...
624 
625  // go left
626  treenode<valuetype> *node=left;
627 
628  // go as far right as possible
629  while (node->getRightChild()) {
630  node=node->getRightChild();
631  }
632  return node;
633 
634  } else if (parent) {
635 
636  // if we're the right child of our parent,
637  // then our parent contains the next lowest value
638  //
639  // Why the weird cast? I really don't know. gcc 2.95.2 on
640  // some platforms (os 10.0, 10.1, and mklinux) require it, but
641  // oddly don't in other cases of comparing left/right child to
642  // this.
643  if ((treenode<valuetype> *)parent->getRightChild()==
644  (treenode<valuetype> *)this) {
645  return parent;
646  }
647 
648  // If we're the left child of our parent, then we have to
649  // move up until we find an acestor that's the right child of
650  // its parent. That node contains the next lowest value.
651  treenode<valuetype> *node=parent;
652  while (node) {
653  if (!node->getParent()) {
654  break;
655  }
656  if (node->getParent()->getRightChild()==node) {
657  return node->getParent();
658  }
659  node=node->getParent();
660  }
661  }
662  return NULL;
663 }

665 template <class valuetype>
666 inline
667 treenode<valuetype> *avltreenode<valuetype>::getNext() {
668 
669  // in-order, depth-first traversal...
670 
671  if (right) {
672 
673  // if we have a right child then its leftmost descendent
674  // contains the next highest value...
675 
676  // go right
677  treenode<valuetype> *node=right;
678 
679  // go as far left as possible
680  while (node->getLeftChild()) {
681  node=node->getLeftChild();
682  }
683  return node;
684 
685  } else if (parent) {
686 
687  // if we're the left child of our parent,
688  // then our parent contains the next highest value
689  //
690  // Why the weird cast? I really don't know. gcc 2.95.2 on
691  // some platforms (os 10.0, 10.1, and mklinux) require it, but
692  // oddly don't in other cases of comparing left/right child to
693  // this.
694  if ((treenode<valuetype> *)parent->getLeftChild()==
695  (treenode<valuetype> *)this) {
696  return parent;
697  }
698 
699  // If we're the right child of our parent, then we have to
700  // move up until we find an acestor that's the left child of
701  // its parent. That node contains the next highest value.
702  treenode<valuetype> *node=parent;
703  while (node) {
704  if (!node->getParent()) {
705  break;
706  }
707  if (node->getParent()->getLeftChild()==node) {
708  return node->getParent();
709  }
710  node=node->getParent();
711  }
712  }
713  return NULL;
714 }

746 template <class valuetype>
747 inline
748 void avltreenode<valuetype>::detach(treenode<valuetype> **treetop) {
749 
750  if (left && right) {
751 
752  // node with left and right children...
753 
754  // get this node's successor...
755  //
756  // (eg. if the tree contains values 5, 7, 10, 12, 15, and 18,
757  // and this node is 10, then find the node with 12 in it)
758  //
759  // following the rules from our in-order, depth-first traversal
760  // above, since we have a right child, we must go right one,
761  // then go left as far as possible
762  treenode<valuetype> *successor=right;
763  while (successor->getLeftChild()) {
764  successor=successor->getLeftChild();
765  }
766 
767  // if the successor was the immediate right child of this node
768  // then we need to handle a few things differently later
769  bool successorwasimmediaterightchild=(right==successor);
770 
771 
772  // swap this node with the successor...
773 
774  // get a copy of the successor
775  avltreenode<valuetype> temp(
776  *((avltreenode<valuetype> *)successor));
777 
778  // re-parent the successor
779  successor->setParent(parent);
780  if (successor->getParent()) {
781  if (successor->getParent()->getLeftChild()==this) {
782  successor->getParent()->
783  setLeftChild(successor);
784  } else {
785  successor->getParent()->
786  setRightChild(successor);
787  }
788  } else {
789  *treetop=successor;
790  }
791 
792  // replace the successor's children
793  // with those of this node
794  successor->setLeftChild(left);
795  if (successor->getLeftChild()) {
796  successor->getLeftChild()->setParent(successor);
797  }

871  }

888  if (child) {

890  }

893  left=NULL;
894  right=NULL;

905 void avltreenode<valuetype>::adjustParentHeights(treenode<valuetype> *node) {
906 
907  // move up the tree, starting with the parent of "node"...
908  while (node->getParent()) {
909 
910  // calculate the new height of the parent
911  uint8_t height=((node->getLeftHeight()>
912  node->getRightHeight())?
913  node->getLeftHeight():
914  node->getRightHeight())+1;
915 
916  // If "node" is the left child of the parent, then adjust the
917  // parent's left height.
918  // If "node" is the right child of the parent, then adjust the
919  // parent's right height.
920  // In either case, bail if the height is already the same as we
921  // calculated.
922  if (node->getParent()->getLeftChild()==node) {
923  if (node->getParent()->getLeftHeight()==height) {
924  return;
925  }
926  node->getParent()->setLeftHeight(height);
927  } else {
928  if (node->getParent()->getRightHeight()==height) {
929  return;
930  }
931  node->getParent()->setRightHeight(height);
932  }
933 
934  // move up
935  node=node->getParent();
936  }

942 
943  // AVL balance...
944 
945  // start balancing with the current node
946  treenode<valuetype> *node=this;
947  while (node) {
948 
949  // there's an imbalance if the left and right
950  // tree heights differ by more than 1
951  if ((node->getLeftHeight()>node->getRightHeight() &&
952  node->getLeftHeight()-node->getRightHeight()>1) ||
953  (node->getRightHeight()>node->getLeftHeight() &&
954  node->getRightHeight()-node->getLeftHeight()>1)) {
955 
956  // apply the appropriate rotation to restore balance
957  // and let the rotation method tell us whch node to
958  // process next
959  if (node->getRightHeight()>node->getLeftHeight()) {
960  if (node->getRightChild()->getRightHeight()>
961  node->getRightChild()->getLeftHeight()) {
962  node=node->leftRotate(treetop);
963  } else {
964  node=node->rightLeftRotate(treetop);
965  }
966  } else if (node->getLeftHeight()>
967  node->getRightHeight()) {
968  if (node->getLeftChild()->getLeftHeight()>
969  node->getLeftChild()->
970  getRightHeight()) {
971  node=node->rightRotate(treetop);
972  } else {
973  node=node->leftRightRotate(treetop);
974  }
975  }
976 
977  } else {
978 
979  // if there's no imbalance then the next node we need
980  // to process is the parent of the current node
981  node=node->getParent();
982  }
983  }
