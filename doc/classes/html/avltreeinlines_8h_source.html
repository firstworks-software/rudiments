137  avltreenode<valuetype> *newnode=new avltreenode<valuetype>(NULL);
138 
139  // copy the value
140  newnode->setValue(node_duplicate_value(node->getValue(),

177 template <class valuetype>
178 inline
179 void avltree<valuetype>::insert(treenode<valuetype> *node) {
180 
181  // degenerate case
182  if (!node) {
183  return;
184  }
185 
186  #ifdef DEBUG_AVLTREE
187  stdoutput.printf("----------------------------------------"
188  "---------------------------------------\n");
189  #endif
190 
191  if (top) {
192 
193  // insert the node, optionally replacing the top of the tree
194  insert(top,node,&top);
195 
196  // update first
197  for (first=top;
198  first->getLeftChild();
199  first=first->getLeftChild()) {}
200 
201  // update last
202  for (last=top;
203  last->getRightChild();
204  last=last->getRightChild()) {}
205  } else {
206 
207  // if there was no top node, then this is the
208  // first node inserted into the entire tree
209  top=node;
210  first=node;
211  last=node;
212  }
213 
214  // increment length
215  length++;
216 }

218 template <class valuetype>
219 inline
220 void avltree<valuetype>::insert(treenode<valuetype> *top,
221  treenode<valuetype> *node,
222  treenode<valuetype> **treetop) {
223 
224  // degenerate case
225  if (!node) {
226  return;
227  }
228 
229  // find a location to insert the node (should always be a leaf node)
230  treenode<valuetype> *location=top;
231  for (;;) {
232 
233  if (this->getComparator()->compare(node->getValue(),
234  location->getValue())<=0) {
235 
236  if (location->getLeftChild()) {
237  location=location->getLeftChild();
238  } else {
239 
240  #ifdef DEBUG_AVLTREE
241  stdoutput.printf("insert ");
242  node_print(node->getValue());
243  stdoutput.printf(" to left of ");
244  node_print(location->getValue());
245  stdoutput.printf(" {\n\n");
246  #endif
247 
248  location->setLeftChild(node);
249  break;
250  }
251 
252  } else if (this->getComparator()->compare(
253  node->getValue(),
254  location->getValue())>0) {
255 
256  if (location->getRightChild()) {
257  location=location->getRightChild();
258  } else {
259 
260  #ifdef DEBUG_AVLTREE
261  stdoutput.printf("insert ");
262  node_print(node->getValue());
263  stdoutput.printf(" to right of ");
264  node_print(location->getValue());
265  stdoutput.printf(" {\n\n");
266  #endif
267 
268  location->setRightChild(node);
269  break;
270  }
271  }
272  }
273 
274  node->setParent(location);
275 
276  // update heights up the tree
277  top->adjustParentHeights(node);
278 
279  // balance the tree
280  node->getParent()->balance(treetop);
281 
282  #ifdef DEBUG_AVLTREE
283  stdoutput.printf("} insert\n\n");
284  #endif
285 }

290 
291  // degenerate case
292  if (!node) {
293  return NULL;
294  }
295 
296  #ifdef DEBUG_AVLTREE
297  stdoutput.printf("----------------------------------------"

302  if (first==node) {
303  first=node->getNext();

305 
306  // update last
307  if (last==node) {
308  last=node->getPrevious();

393 inline
394 treenode<valuetype> *avltree<valuetype>::find(
395  treenode<valuetype> *startnode,
396  valuetype value) const {
397 
398  #ifdef DEBUG_AVLTREE
399  stdoutput.printf("find ");
400  node_print(value);
401  stdoutput.printf(" from ");
402  if (startnode) {
403  node_print(startnode->getValue());
404  } else {
405  node_print("(null)");
406  }
407  stdoutput.printf(" {\n");
408  #endif
409 
410  // descend the tree until we find the value or run off of the bottom
411  treenode<valuetype> *current=startnode;
412  while (current) {
413 
414  int32_t result=this->getComparator()->compare(
415  current->getValue(),value);
416 
417  #ifdef DEBUG_AVLTREE
418  stdoutput.printf(" ");
419  node_print(current->getValue());
420  stdoutput.printf(" - %d\n",result);
421  #endif
422 
423  if (result<0) {
424  current=current->getRightChild();
425  } else if (result==0) {
426  break;
427  } else if (result>0) {
428  current=current->getLeftChild();
429  }
430  }
431 
432  #ifdef DEBUG_AVLTREE
433  if (current) {
434  stdoutput.printf(" success!\n");
435  } else {
436  stdoutput.printf(" failed\n");
437  }
438  stdoutput.printf("} find\n\n");
439  #endif
440 
441  return current;
442 }

452 template <class valuetype>
453 inline
454 void avltree<valuetype>::clear() {
455 
456  #ifdef DEBUG_AVLTREE
457  uint64_t i=0;
458  stdoutput.printf("clearing %d nodes (%08x) {\n",length,this);
459  #endif
460 
461  // start at the top
462  treenode<valuetype> *node=top;
463  while (node) {
464 
465  // go right one, then go left as far as possible
466  if (node->getRightChild()) {
467  node=node->getRightChild();
468  }
469  while (node->getLeftChild()) {
470  node=node->getLeftChild();
471  }
472 
473  // get the parent
474  treenode<valuetype> *p=node->getParent();
475  if (p) {
476  if (p->getLeftChild()==node) {
477  p->setLeftChild(NULL);
478  } else {
479  p->setRightChild(NULL);
480  }
481  }
482 
483  // delete the node
484  #ifdef DEBUG_AVLTREE
485  stdoutput.printf(" clearing %lld: (%08x) ",
486  i,node->getValue());
487  node_print(node->getValue());
488  stdoutput.printf("\n");
489  i++;
490  #endif
491  node_delete_value(node->getValue(),
492  this->getManageValues(),
493  this->getManageArrayValues());
494  delete node;
495 
496  // continue with parent...
497  node=p;
498  }
499 
500  #ifdef DEBUG_AVLTREE
501  stdoutput.printf("} cleared %d nodes\n\n",i);
502  #endif
503 
504  // clear pointers and length
505  top=NULL;
506  first=NULL;
507  last=NULL;
508  length=0;
509 }

576 template <class valuetype>
577 inline
578 treenode<valuetype> *avltreenode<valuetype>::getPrevious() const {
579 
580  // reverse in-order, depth-first traversal...
581 
582  if (left) {
583 
584  // if we have a left child then its rightmost descendent
585  // contains the next lowest value...
586 
587  // go left
588  treenode<valuetype> *node=left;
589 
590  // go as far right as possible
591  while (node->getRightChild()) {
592  node=node->getRightChild();
593  }
594  return node;
595 
596  } else if (parent) {
597 
598  // if we're the right child of our parent,
599  // then our parent contains the next lowest value
600  //
601  // Why the weird cast? I really don't know. gcc 2.95.2 on
602  // some platforms (os 10.0, 10.1, and mklinux) require it, but
603  // oddly don't in other cases of comparing left/right child to
604  // this.
605  if ((const treenode<valuetype> *)parent->getRightChild()==
606  (const treenode<valuetype> *)this) {
607  return parent;
608  }
609 
610  // If we're the left child of our parent, then we have to
611  // move up until we find an acestor that's the right child of
612  // its parent. That node contains the next lowest value.
613  treenode<valuetype> *node=parent;
614  while (node) {
615  if (!node->getParent()) {
616  break;
617  }
618  if (node->getParent()->getRightChild()==node) {
619  return node->getParent();
620  }
621  node=node->getParent();
622  }
623  }
624  return NULL;
625 }

628 inline
629 treenode<valuetype> *avltreenode<valuetype>::getNext() const {
630 
631  // in-order, depth-first traversal...
632 
633  if (right) {
634 
635  // if we have a right child then its leftmost descendent
636  // contains the next highest value...
637 
638  // go right
639  treenode<valuetype> *node=right;
640 
641  // go as far left as possible
642  while (node->getLeftChild()) {
643  node=node->getLeftChild();
644  }
645  return node;
646 
647  } else if (parent) {
648 
649  // if we're the left child of our parent,
650  // then our parent contains the next highest value
651  //
652  // Why the weird cast? I really don't know. gcc 2.95.2 on
653  // some platforms (os 10.0, 10.1, and mklinux) require it, but
654  // oddly don't in other cases of comparing left/right child to
655  // this.
656  if ((const treenode<valuetype> *)parent->getLeftChild()==
657  (const treenode<valuetype> *)this) {
658  return parent;
659  }
660 
661  // If we're the right child of our parent, then we have to
662  // move up until we find an acestor that's the left child of
663  // its parent. That node contains the next highest value.
664  treenode<valuetype> *node=parent;
665  while (node) {
666  if (!node->getParent()) {
667  break;
668  }
669  if (node->getParent()->getLeftChild()==node) {
670  return node->getParent();
671  }
672  node=node->getParent();
673  }
674  }

748 template <class valuetype>
749 inline
750 void avltreenode<valuetype>::detach(treenode<valuetype> **treetop) {
751 
752  #ifdef DEBUG_AVLTREE
753  stdoutput.printf("detach ");
754  node_print(value);
755  stdoutput.printf(" {\n\n");
756 
757  treenode<valuetype> *top=this;
758  while (top->getParent()) { top=top->getParent(); }
759  node_print(top->getValue()); stdoutput.printf("\n");
760  #endif
761 
762  if (left && right) {
763 
764  // node with left and right children...
765 
766  #ifdef DEBUG_AVLTREE
767  stdoutput.printf("less simple case: 2 children\n\n");
768  #endif
769 
770  // get this node's successor...
771  //
772  // (eg. if the tree contains values 5, 7, 10, 12, 15, and 18,
773  // and this node is 10, then find the node with 12 in it)
774  //
775  // following the rules from our in-order, depth-first traversal
776  // above, since we have a right child, we must go right one,
777  // then go left as far as possible
778  treenode<valuetype> *successor=right;
779  while (successor->getLeftChild()) {
780  successor=successor->getLeftChild();
781  }
782 
783  #ifdef DEBUG_AVLTREE
784  stdoutput.printf("swap ");
785  node_print(value);
786  stdoutput.printf(" and ");
787  node_print(successor->getValue());
788  stdoutput.printf("\n\n");
789  #endif
790 
791 
792  // if the successor was the immediate right child of this node
793  // then we need to handle a few things differently later
794  bool successorwasimmediaterightchild=(right==successor);
795 
796 
797  // swap this node with the successor...
798 
799  // get a copy of the successor
800  avltreenode<valuetype> temp(
801  *((avltreenode<valuetype> *)successor));
802 
803  // re-parent the successor
804  successor->setParent(parent);
805  if (successor->getParent()) {
806  if (successor->getParent()->getLeftChild()==this) {
807  successor->getParent()->
808  setLeftChild(successor);
809  } else {
810  successor->getParent()->
811  setRightChild(successor);
812  }
813  } else {
814  *treetop=successor;
815  }
816 
817  // replace the successor's children
818  // with those of this node
819  successor->setLeftChild(left);
820  if (successor->getLeftChild()) {
821  successor->getLeftChild()->setParent(successor);
822  }
823  if (successorwasimmediaterightchild) {
824  successor->setRightChild(this);

950 void avltreenode<valuetype>::adjustParentHeights(treenode<valuetype> *node) {
951 
952  // move up the tree, starting with the parent of "node"...
953  while (node->getParent()) {
954 
955  // calculate the new height of the parent
956  uint8_t height=((node->getLeftHeight()>

959  node->getRightHeight())+1;

962  // parent's left height.
963  // If "node" is the right child of the parent, then adjust the
964  // parent's right height.
965  // In either case, bail if the height is already the same as we
966  // calculated.
967  if (node->getParent()->getLeftChild()==node) {
968  if (node->getParent()->getLeftHeight()==height) {
969  return;

972  } else {
973  if (node->getParent()->getRightHeight()==height) {
974  return;
975  }
976  node->getParent()->setRightHeight(height);
